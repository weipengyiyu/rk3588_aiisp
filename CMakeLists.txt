cmake_minimum_required(VERSION 3.16)
project(rk3588_aiisp VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." CACHE PATH "RK3588 SDK root")
set(RKNN_INCLUDE_DIR "${SDK_ROOT}/external/rknpu2/runtime/Linux/librknn_api/include")
if(NOT EXISTS "${RKNN_INCLUDE_DIR}/rknn_api.h")
  message(FATAL_ERROR "rknn_api.h not found; set SDK_ROOT to the existing RK3588 SDK")
endif()
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
add_library(raw_core src/raw.cpp)
target_include_directories(raw_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include/rk3588-aiisp>)
add_library(aiisp_core src/preprocess.cpp src/npu.cpp src/postprocess.cpp src/pipeline.cpp)
target_include_directories(aiisp_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include/rk3588-aiisp> PRIVATE "${RKNN_INCLUDE_DIR}")
target_link_libraries(aiisp_core PUBLIC raw_core PRIVATE dl)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
  target_compile_options(aiisp_core PRIVATE -march=armv8.2-a+fp16)
endif()
add_library(aiisp_v4l2 src/video.cpp src/isp.cpp)
target_link_libraries(aiisp_v4l2 PUBLIC aiisp_core)
add_executable(rk3588_aiisp src/main.cpp)
target_link_libraries(rk3588_aiisp PRIVATE aiisp_v4l2)
include(CTest)
if(BUILD_TESTING)
  add_executable(test_raw tests/test_raw.cpp)
  target_link_libraries(test_raw PRIVATE raw_core)
  add_executable(test_pipeline tests/test_pipeline.cpp)
  target_link_libraries(test_pipeline PRIVATE aiisp_core)
  add_test(NAME raw_contract COMMAND test_raw)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME pipeline_contract COMMAND test_pipeline)
  add_test(NAME launcher_contract COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_launcher.py")
  add_test(NAME clean_contract COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_clean.py")
  if(NOT CMAKE_CROSSCOMPILING)
    add_test(NAME offline_contract COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_offline.py" $<TARGET_FILE:rk3588_aiisp>)
  endif()
endif()
install(TARGETS raw_core aiisp_core aiisp_v4l2 rk3588_aiisp
        EXPORT rk3588AiispTargets ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include/rk3588-aiisp)
install(EXPORT rk3588AiispTargets NAMESPACE rk3588:: DESTINATION lib/cmake/rk3588-aiisp)
install(FILES cmake/rk3588-aiispConfig.cmake DESTINATION lib/cmake/rk3588-aiisp)
